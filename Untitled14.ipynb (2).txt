        "def parse_config(config_file_path):\n",
        "    \"\"\"\n",
        "    Legge un file di testo riga per riga e crea un dizionario {chiave: valore}.\n",
        "    Ignora le righe vuote o quelle che iniziano con '#'.\n",
        "    \"\"\"\n",
        "    config = {}\n",
        "    with open(config_file_path, 'r', encoding='utf-8') as f:\n",
        "        for line in f:\n",
        "            line = line.strip()\n",
        "            if not line or line.startswith('#'):\n",
        "                continue\n",
        "            if '=' in line:\n",
        "                key, value = line.split('=', 1)\n",
        "                config[key.strip()] = value.strip()\n",
        "    return config\n",
        "\n",
        "def get_outlook_attachment(outlook_folder_name):\n",
        "    \"\"\"\n",
        "    - Apre Outlook\n",
        "    - Cerca la cartella `outlook_folder_name`\n",
        "    - Restituisce il percorso del file Excel salvato localmente (attachment)\n",
        "      se lo trova. Altrimenti None.\n",
        "\n",
        "    Si assume che ci sia UNA SOLA email in quella cartella e che contenga UN SOLO\n",
        "    file Excel in allegato.\n",
        "    \"\"\"\n",
        "    outlook = win32.Dispatch(\"Outlook.Application\").GetNamespace(\"MAPI\")\n",
        "\n",
        "    # Se vuoi accedere alla cartella \"default\" (ad es. Posta in arrivo) di un\n",
        "    # determinato account, puoi dover fare:\n",
        "    #   account = outlook.Folders[\"IlTuoIndirizzoEmail\"]\n",
        "    #   folder = account.Folders[outlook_folder_name]\n",
        "    #\n",
        "    # In molti casi, la cartella potrebbe trovarsi direttamente sotto il \"Default Store\".\n",
        "    # Per semplicità si prova con:\n",
        "    folder = outlook.Folders.Item(1).Folders[outlook_folder_name]\n",
        "\n",
        "    items = folder.Items\n",
        "    # Si ipotizza che ci sia una sola mail\n",
        "    if len(items) == 0:\n",
        "        print(\"Nessuna email trovata nella cartella:\", outlook_folder_name)\n",
        "        return None\n",
        "\n",
        "    mail = items[0]  # la prima (o unica) email nella cartella\n",
        "    attachments = mail.Attachments\n",
        "    if attachments.Count == 0:\n",

        "        return None\n",

        "    # Supponiamo che esista un solo allegato Excel\n",

        "    # Salviamo localmente\n",
        "    temp_path = os.path.join(os.getcwd(), attachment.FileName)\n",
        "    attachment.SaveAsFile(temp_path)\n",
        "\n",
        "    return temp_path\n",
